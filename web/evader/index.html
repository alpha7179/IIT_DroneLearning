  + w_shaping  * (d_prev - d_now)
  + w_survival * 1
  + w_occlusion * R_los
  + w_time     * (-1)
  + [Goal Reached] +1.0
  + [Captured]     -1.0
  + [Timeout]      +0.2
  + [Crash]        -1.0
